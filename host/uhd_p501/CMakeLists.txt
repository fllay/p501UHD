# uhd_p501 -- UHD 4.6 out-of-tree module for the PZSDR P501.
#   cmake -B build -DUHD_ROOT=/usr/local && cmake --build build -j
#   UHD_MODULE_PATH=$PWD/build/libuhd_p501.so uhd_find_devices
cmake_minimum_required(VERSION 3.16)
project(uhd_p501 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(UHD_ROOT "/usr/local" CACHE PATH "Prefix of the UHD 4.6 install the apps use")
find_library(UHD_LIBRARIES NAMES uhd PATHS ${UHD_ROOT}/lib ${UHD_ROOT}/lib64 ${UHD_ROOT}/lib/${CMAKE_LIBRARY_ARCHITECTURE} NO_DEFAULT_PATH)
find_path(UHD_INCLUDE_DIRS NAMES uhd/version.hpp PATHS ${UHD_ROOT}/include NO_DEFAULT_PATH)
if(NOT UHD_LIBRARIES OR NOT UHD_INCLUDE_DIRS)
    message(FATAL_ERROR "UHD not found under ${UHD_ROOT}")
endif()
get_filename_component(_real "${UHD_LIBRARIES}" REALPATH)
message(STATUS "uhd_p501: headers ${UHD_INCLUDE_DIRS}, library ${_real}")
if(NOT _real MATCHES "libuhd\\.so\\.4\\.6")
    message(FATAL_ERROR "uhd_p501 must be built against UHD 4.6 (got ${_real})")
endif()
find_package(Threads REQUIRED)
find_package(Boost REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED libusb-1.0)
add_library(uhd_p501 MODULE p501_device.cpp)
# make <uhd/...> resolve to the pinned headers
set(_shim "${CMAKE_CURRENT_BINARY_DIR}/uhd_include")
file(MAKE_DIRECTORY "${_shim}")
file(CREATE_LINK "${UHD_INCLUDE_DIRS}/uhd" "${_shim}/uhd" SYMBOLIC)
target_include_directories(uhd_p501 BEFORE PRIVATE "${_shim}")
target_include_directories(uhd_p501 PRIVATE ${Boost_INCLUDE_DIRS} ${LIBUSB_INCLUDE_DIRS})
target_link_libraries(uhd_p501 PRIVATE ${UHD_LIBRARIES} ${LIBUSB_LIBRARIES} Threads::Threads)
target_compile_options(uhd_p501 PRIVATE -O3 -Wall -Wextra -Wno-unused-parameter)
target_link_options(uhd_p501 PRIVATE -Wl,--no-undefined)
set_target_properties(uhd_p501 PROPERTIES PREFIX "lib")
